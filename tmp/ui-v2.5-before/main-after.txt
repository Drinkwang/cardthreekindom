func _ready() -> void:
	theme = Art.theme()
	_build()
	GameState.changed.connect(_on_changed)
	GameState.map_changed.connect(_on_map_changed)
	GameState.shop_changed.connect(_refresh_shop)
	GameState.message.connect(_append_log)
	_on_map_changed()
	_refresh_shop()
	_on_changed()
	if not GameState.in_battle and not GameState.last_outcome.is_empty():
		_open_home()
	if GameState.offline_report != "" and not _offline_shown:
		_offline_shown = true
		_append_log(GameState.offline_report)
	_maybe_story.call_deferred()


func _process(delta: float) -> void:
	# 看升级与札记时暂停手动轮；自动经营仍按自己的节奏运行。
	var overlay_open := (_home != null and _home.visible) or (_map_layer != null and _map_layer.visible) \
		or (_city != null and _city.visible) or (_shop != null and _shop.visible) \
		or (_story != null and _story.visible) or (_reveal != null and _reveal.visible) or _drawer_open or _pool_open
	GameState.narrative_paused = overlay_open and not GameState.automation_enabled
	if _camera_toolbar != null: _camera_toolbar.visible = not overlay_open
	if _minimap != null: _minimap.visible = not overlay_open
	if _table != null and _slap_cursor != null:
		var pointer := _cursor_preview if _cursor_preview.x >= 0 else _table.get_local_mouse_position()
		if _window_focused: _advance_edge_pan(delta, pointer)
		_update_slap_cursor(pointer)
	_round_ui_clock += delta
	if _round_ui_clock >= 0.05:
		_round_ui_clock = 0.0
		_update_topbar()
		_update_minimap()

func _input(event: InputEvent) -> void:
	if event is InputEventMouseButton and event.button_index == MOUSE_BUTTON_MIDDLE and not event.pressed:
		_panning = false
	if event is InputEventKey and event.pressed and not event.echo and event.keycode == KEY_M and _input_allowed():
		_minimap.toggle()
		get_viewport().set_input_as_handled()

func _notification(what: int) -> void:
	if what == NOTIFICATION_WM_WINDOW_FOCUS_OUT:
		_window_focused = false
		_panning = false
	elif what == NOTIFICATION_WM_WINDOW_FOCUS_IN:
		_window_focused = true

func _build() -> void:
	set_anchors_preset(Control.PRESET_FULL_RECT)

	# S5：整棵子树的按钮统一换成印刷纸按钮（已在按钮上加过 override 的除外）
	var th := _paper_button_theme()
	if th != null:
		theme = th

	var root := VBoxContainer.new()
	root.set_anchors_preset(Control.PRESET_FULL_RECT)
	root.add_theme_constant_override("separation", 0)
	add_child(root)

	root.add_child(_build_topbar())

	var mid := HBoxContainer.new()
	mid.size_flags_vertical = Control.SIZE_EXPAND_FILL
	mid.add_theme_constant_override("separation", 0)
	mid.add_child(_build_table())
	_drawer = _build_drawer()
	mid.add_child(_drawer)
	root.add_child(mid)

	_pool = _build_pool()
	root.add_child(_pool)
	root.add_child(_build_bottombar())

	# 覆盖层：舆图 / 大本营 / 城建（在下），翻牌揭示（最上）
	_map_layer = _build_map_layer()
	add_child(_map_layer)
	_home = PaanHome.new()
	_home.deploy.connect(_on_home_deploy)
	_home.request_map.connect(_on_home_map)
	_home.request_city.connect(_open_city)
	_home.request_shop.connect(_open_shop)
	_home.closed.connect(_close_home)
	_home.practice_requested.connect(_on_home_practice)
	_home.request_journal.connect(_open_journal)
	add_child(_home)
	_home.visible = false
	_city = PaanCity.new()
	_city.closed.connect(_return_from_city)
	_city.request_home.connect(_open_home)
	_city.request_map.connect(_on_home_map)
	_city.deploy.connect(_on_city_deploy)
	add_child(_city)
	_city.visible = false
	_shop = PaanShop.new()
	_shop.closed.connect(_return_from_shop)
	add_child(_shop)
	_shop.visible = false
	_reveal = PaanReveal.new()
	add_child(_reveal)

func _world_to_screen(point: Vector2) -> Vector2:
	return _pile_layer.position + point * _camera_scale

func _screen_to_world(point: Vector2) -> Vector2:
	return (point - _pile_layer.position) / maxf(0.0001, _camera_scale)

func _apply_camera() -> void:
	if _pile_layer == null or _table.size.y < 10: return
	_camera_scale = minf(_table.size.x / _world_size.x, _table.size.y / _world_size.y) * _camera_zoom
	var limit := ((_world_size * _camera_scale - _table.size) * 0.5).max(Vector2.ZERO)
	_camera_pan.x = clampf(_camera_pan.x, -limit.x, limit.x)
	_camera_pan.y = clampf(_camera_pan.y, -limit.y, limit.y)
	_pile_layer.scale = Vector2.ONE * _camera_scale
	_pile_layer.position = (_table.size - _world_size * _camera_scale) * 0.5 + _camera_pan
	if _zoom_label != null: _zoom_label.text = "%d%%" % roundi(_camera_zoom * 100.0)
	if _camera_toolbar != null:
		_camera_toolbar.position = Vector2(16, 14)
	if _minimap != null:
		_minimap.position = Vector2(maxf(8, _table.size.x - _minimap.size.x - 16), 14)
	_update_minimap()

func _viewport_world_rect() -> Rect2:
	return Rect2(_screen_to_world(Vector2.ZERO), _table.size / maxf(_camera_scale, 0.0001)).intersection(Rect2(Vector2.ZERO, _world_size))

func _update_minimap() -> void:
	if _minimap == null or _pile_layer == null: return
	var positions: Array[Vector2] = []
	for i in range(mini(_cards.size(), GameState.battle.size())):
		var card: Dictionary = _cards[i]
		if not card.has("rest_pos"): continue
		if not bool(card.dead) and float(GameState.battle[i].hp) > 0:
			positions.append(card.rest_pos + card.size * 0.5)
	_minimap.update_view(_viewport_world_rect(), positions)
	_minimap.set_city_title(str(GameData.region(GameState.battle_region).get("name", "城池")))

func _over_camera_ui(screen_pos: Vector2) -> bool:
	return (_camera_toolbar != null and _camera_toolbar.visible and _camera_toolbar.get_rect().has_point(screen_pos)) \
		or (_minimap != null and _minimap.visible and _minimap.get_rect().has_point(screen_pos))

func _advance_edge_pan(delta: float, screen_pos: Vector2) -> void:
	if not _input_allowed() or _panning or _over_camera_ui(screen_pos) or not Rect2(Vector2.ZERO, _table.size).has_point(screen_pos): return
	var direction := Vector2.ZERO
	if screen_pos.x < EDGE_PAN_BAND: direction.x = 1.0 - screen_pos.x / EDGE_PAN_BAND
	elif screen_pos.x > _table.size.x - EDGE_PAN_BAND: direction.x = -1.0 + (_table.size.x - screen_pos.x) / EDGE_PAN_BAND
	if screen_pos.y < EDGE_PAN_BAND: direction.y = 1.0 - screen_pos.y / EDGE_PAN_BAND
	elif screen_pos.y > _table.size.y - EDGE_PAN_BAND: direction.y = -1.0 + (_table.size.y - screen_pos.y) / EDGE_PAN_BAND
	if direction.length_squared() <= 0.0001: return
	if direction.length_squared() > 1.0: direction = direction.normalized()
	_camera_pan += direction * EDGE_PAN_SPEED * minf(delta, 0.05)
	_apply_camera()

func _navigate_minimap(world_point: Vector2) -> void:
	if not _input_allowed(): return
	_camera_pan = (_world_size * 0.5 - world_point) * _camera_scale
	_apply_camera()
	_update_slap_cursor(_table.get_local_mouse_position())

func _set_camera_zoom(value: float, screen_anchor: Vector2) -> void:
	var anchor := _screen_to_world(screen_anchor)
	_camera_zoom = clampf(value, ZOOM_MIN, ZOOM_MAX)
	_apply_camera()
	_camera_pan += screen_anchor - _world_to_screen(anchor)
	_apply_camera()
	_update_slap_cursor(screen_anchor)

func _reset_camera() -> void:
	_camera_zoom = 1.0
	_camera_pan = Vector2.ZERO
	_apply_camera()

func _focus_starting_district() -> void:
	_camera_zoom = DEFAULT_ZOOM
	_apply_camera()
	_camera_pan = (_world_size * 0.5 - _world_size * Vector2(0.42, 0.32)) * _camera_scale
	_apply_camera()

func _input_allowed() -> bool:
	if not GameState.in_battle or not GameState.round_active or GameState.narrative_paused: return false
	return not ((_home != null and _home.visible) or (_map_layer != null and _map_layer.visible)
		or (_city != null and _city.visible) or (_shop != null and _shop.visible)
		or (_story != null and _story.visible) or (_reveal != null and _reveal.visible)
		or _drawer_open or _pool_open)

func _update_slap_cursor(screen_pos: Vector2) -> void:
	var over_tools := _over_camera_ui(screen_pos)
	var active := _window_focused and _input_allowed() and Rect2(Vector2.ZERO, _table.size).has_point(screen_pos) and not over_tools and not _panning
	_slap_cursor.visible = active
	if active != _cursor_hidden:
		_cursor_hidden = active
		Input.set_mouse_mode(Input.MOUSE_MODE_HIDDEN if active else Input.MOUSE_MODE_VISIBLE)
	if not active: return
	_slap_cursor.center = screen_pos
	_slap_cursor.radius = GameState.slap_radius() * _camera_scale
	_slap_cursor.cooldown = clampf(GameState.slap_cooldown_left / GameState.slap_interval(), 0, 1)
	var world := _screen_to_world(screen_pos)
	var hits := _area_targets(world, GameState.slap_radius())
	_slap_cursor.targets = hits.size()
	_slap_cursor.outlines = []
	for i in hits:
		var card: Dictionary = _cards[i]
		var center: Vector2 = card.rest_pos + card.size * 0.5
		var corners := PackedVector2Array()
		for corner in [Vector2(-1, -1), Vector2(1, -1), Vector2(1, 1), Vector2(-1, 1)]:
			corners.append(_world_to_screen(center + (card.size * corner * 0.5).rotated(float(card.rot))))
		_slap_cursor.outlines.append(corners)
	_slap_cursor.queue_redraw()

# 底部上阵条：槽位 + 卡池（v1.0）
# =====================================================================
## 由 _on_changed() 调用。用一个"指纹"串判断要不要重建 —— 拍卡时 changed 每一下都发，
## 每次都重建 7 个槽位 + 几十张卡池牌会白烧 CPU（也让人眼看到闪烁）。
func _refresh_deploy_bar() -> void:
	var sig := "%d|%d" % [GameState.carry.size(), GameState.carry_max()]
	for id in GameState.carry:
		var sid := str(id)
		sig += "|" + sid
		# v1.1：装备/兵的数量也要进指纹 —— 不然在大本营挂了装备回来，槽位上还是旧数字
		sig += ":%d:%d" % [GameState.hero_equip_of(sid).size(), GameState.hero_troops_of(sid).size()]
	if sig != _bar_sig:
		_bar_sig = sig
		_rebuild_slots()
		_pool_sig = ""                 # 上阵名单变了 → 卡池里的「待命」集合也变了
	if _pool_open:
		_rebuild_pool()


func _rebuild_slots() -> void:
	_clear(_slot_row)
	var cap := GameState.carry_max()
	_lbl_bar.text = "%d/%d" % [GameState.carry.size(), cap]
	# 主角固定位：永远在最左，不占携带位、点不掉、拖不进（铁律 5：主角固定不占位不可卸）
	_slot_row.add_child(_mk_slot_node(HERO_ALWAYS, -1, true))
	for i in range(cap):
		var id := str(GameState.carry[i]) if i < GameState.carry.size() else ""
		_slot_row.add_child(_mk_slot_node(id, i, false))


## 战力显示。
## ⚠️ 不能直接用 GameState.fmt() —— 它是给金币用的（四舍五入到整数），
##    而士兵战力是 0.2 / 0.6 / 2.0 这种小数：0.2 会显示成 "0"，看起来像废牌，
##    玩家会以为「士卒」根本没用（用户明确要求要能看清自己带的将兵）。
##    所以战力 < 10 时保留一位小数。
func _fmt_pow(v: float) -> String:
	if absf(v) < 10.0:
		return "%.1f" % v
	return GameState.fmt(v)

func _pool_signature() -> String:
	var ids := []
	for id in GameState.owned.keys():
		var sid := str(id)
		# ⚠️ v1.1：挂在武将麾下带兵的士卒**不进卡池** —— 它们已经"在用"了，
		#    拖也拖不动（carry_add 会拒收），留在池里只会让人以为点了没反应。
		if not GameState.is_carryable(sid) or int(GameState.owned.get(sid, 0)) <= 0 \
				or GameState.in_troops(sid):
			continue
		ids.append("%s:%d:%d" % [sid, int(GameState.owned.get(sid, 0)), GameState.hero_level(sid)])
	ids.sort()
	return "|".join(ids) + "||" + "|".join(GameState.carry)


func _rebuild_pool() -> void:
	var sig := _pool_signature()
	if sig == _pool_sig:
		return
	_pool_sig = sig
	_clear(_pool_box)

	var ids := []
	for id in GameState.owned.keys():
		var sid := str(id)
		if GameState.is_carryable(sid) and int(GameState.owned.get(sid, 0)) > 0 \
				and not GameState.in_carry(sid) and not GameState.in_troops(sid):
			ids.append(sid)

	if ids.is_empty():
		var tip := "卡池里没有待命的将兵 —— 去「商店 → 卡包」开几包，或把上阵的槽位点掉。"
		_pool_box.add_child(_mk_label(tip, 12, INK_DIM))
		return

	# 战力从高到低，一眼看出谁是主力
	ids.sort_custom(func(a: String, b: String) -> bool:
		return GameState.hero_card_power(a) > GameState.hero_card_power(b))
	for id in ids:
		_pool_box.add_child(_mk_pool_card(id))

## 拖过来的牌落到第 index 个槽位。落点已有人 = 换人（旧的回卡池）。
func _deploy_at(id: String, index: int) -> void:
	var cap := GameState.carry_max()
	if index < 0 or index >= cap:
		GameState.carry_add(id)                  # 兜底：当追加处理
		return
	var occupant := str(GameState.carry[index]) if index < GameState.carry.size() else ""
	if occupant != "" and occupant != id:
		GameState.carry_remove(occupant)
	GameState.carry_put(id, index)


func _undeploy(id: String) -> void:
	if GameState.carry_remove(id):
		_pool_sig = ""                            # 有人退回卡池 → 卡池要重画

# ── S5 纸质感资源加载 ────────────────────────────────────────────────
var _tex_cache := {}


func _tex(path: String) -> Texture2D:
	if _tex_cache.has(path):
		return _tex_cache[path]
	var t: Texture2D = null
	if ResourceLoader.exists(path):
		t = load(path) as Texture2D
	_tex_cache[path] = t
	return t


## 纸质感九宫格。margin 必须**等于源纹理上的真实边框宽度**。
## ⚠️ Godot 4 的 StyleBoxTexture **没有 texture_scale**，九宫格的角是按源纹理 1:1 画的，
##    想改变边框粗细只能**重新生成对应尺寸的纹理**，不能在运行时缩放。
## margin 若小于真实边框，卡框内框线会被当成"中心"拉伸糊掉。
## 找不到纹理时返回 null —— 调用方负责回退到 _flat()，这样没图也能跑。
func _paper(_path: String, _margin: float, _scale: float = 1.0) -> StyleBox:
	return Art.panel(Art.PAPER, 8)


func _paper_or_flat(path: String, margin: float, scale: float,
		bg: Color, border: Color, w: int, radius: int) -> StyleBox:
	var sb := _paper(path, margin, scale)
	if sb != null:
		return sb
	return _flat(bg, border, w, radius)


## 全幅背景贴图（牌桌 / 舆图等），等比铺满
func _bg_rect(name: String) -> TextureRect:
	var bg := TextureRect.new()
	bg.texture = _tex(BG_DIR + name)
	bg.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
	bg.stretch_mode = TextureRect.STRETCH_KEEP_ASPECT_COVERED
	bg.mouse_filter = Control.MOUSE_FILTER_IGNORE
	return bg

func _layout_cards() -> void:
	var n := _cards.size()
	_world_size = CityLayout.world_size(n)
	if _pile_layer != null: _pile_layer.size = _world_size
	if _minimap != null: _minimap.set_world(load("res://assets/art_v12/city_world.png"), _world_size)
	_apply_camera()
	var area := _table.size
	if area.x <= 60.0 or area.y <= 60.0 or _cards.is_empty():
		return
	var sz := CARD_SIZE
	var k := clampf(sz.x / CARD_SIZE.x, 0.65, 1.65)
	for i in range(_cards.size()):
		var card: Dictionary = _cards[i]
		card["size"] = sz
		var node: Control = card["node"]
		node.size = sz
		node.pivot_offset = sz * 0.5
		for lbl in card["fonts"].keys():
			(lbl as Label).add_theme_font_size_override("font_size", maxi(8, roundi(float(card["fonts"][lbl]) * k)))
		var margin := 7.0 * k
		card["seal"].position = Vector2(margin, 6 * k)
		card["seal"].size = Vector2(20, 25) * k
		card["top"].position = Vector2(margin + 25 * k, 4 * k)
		card["top"].size = Vector2(sz.x - margin * 2 - 25 * k, 18 * k)
		card["star"].position = Vector2(margin + 25 * k, 21 * k)
		card["star"].size = Vector2(sz.x - margin * 2 - 25 * k, 11 * k)
		card["portrait"].position = Vector2(margin, 34 * k)
		card["portrait"].size = Vector2(sz.x - margin * 2, maxf(20, sz.y - 58 * k))
		card["status"].position = Vector2(margin + 3 * k, 36 * k)
		card["status"].size = Vector2(sz.x - margin * 2 - 6 * k, 20 * k)
		card["status"].add_theme_font_size_override("font_size", maxi(11, roundi(12 * k)))
		card["type"].position = Vector2(margin, sz.y - 19 * k)
		card["type"].size = Vector2(29 * k, 11 * k)
		card["bar"].position = Vector2(margin, sz.y - 22 * k)
		card["bar"].size = Vector2(sz.x - margin * 2, 5 * k)
		card["hpt"].position = Vector2(margin + 30 * k, sz.y - 16 * k)
		card["hpt"].size = Vector2(sz.x - margin * 2 - 30 * k, 11 * k)
		var enemy: Dictionary = GameState.battle[i]
		card["rest_pos"] = Vector2(250, 250) + Vector2(float(enemy.nx), float(enemy.ny)) * (_world_size - Vector2(500, 500)) - sz * 0.5
		if not card["dead"]:
			node.position = card["rest_pos"]
	if _initial_camera:
		_initial_camera = false
		_focus_starting_district()
	_apply_camera()


func _update_card(i: int) -> void:
	if i < 0 or i >= _cards.size() or i >= GameState.battle.size():
		return
	var card: Dictionary = _cards[i]
	if card["dead"]:
		return
	var e: Dictionary = GameState.battle[i]
	var hp := maxf(0.0, float(e["hp"]))
	var hp_max := maxf(1.0, float(e["hp_max"]))
	var bar: ProgressBar = card["bar"]
	bar.max_value = hp_max
	bar.value = hp
	var hpt: Label = card["hpt"]
	hpt.text = "%s/%s" % [_compact_hp(hp), _compact_hp(hp_max)]
	var states := PackedStringArray()
	if int(e.get("thunder", 0)) > 0:
		states.append("雷%d" % int(e["thunder"]))
	if float(e.get("burn_remaining", 0.0)) > 0.0:
		states.append("燃")
	if float(e.get("mark_remaining", 0.0)) > 0.0:
		states.append("标")
	card["status"].text = " · ".join(states)
	card["status"].visible = not states.is_empty()
	if hp <= 0.0:
		_flip_card(i)


func _present_build_effects() -> void:
	var info: Dictionary = GameState.last_slap
	var sequence := int(info.get("effect_seq", -1))
	if sequence < 0 or sequence == _shown_effect_seq:
		return
	_shown_effect_seq = sequence
	if not GameState.in_battle or _table == null:
		return
	if int(info.get("battle_gen", GameState.battle_gen)) != GameState.battle_gen:
		return
	var effects: Array = info.get("effects", [])
	for effect in effects:
		var kind := str(effect.get("kind", ""))
		if kind in ["轻拍", "重拍"]:
			continue
		var index := int(effect.get("index", -1))
		if index < 0 or index >= _cards.size():
			continue
		var card: Dictionary = _cards[index]
		var pos: Vector2 = card.get("rest_pos", card["node"].position) + card["size"] * Vector2(0.5, 0.55)
		var damage := float(effect.get("damage", 0.0))
		var caption := kind + (" -%s" % _fmt_pow(damage) if damage > 0.0 else "")
		var color := Color("294a80") if kind in ["雷爆", "雷印"] else Art.RED
		if kind in ["标记", "白羽追击"]:
			color = Color("21665b")
		var label := Art.label(caption, 16, color)
		label.mouse_filter = Control.MOUSE_FILTER_IGNORE
		label.add_theme_color_override("font_outline_color", Art.PAPER_LIGHT)
		label.add_theme_constant_override("outline_size", 3)
		label.position = pos + Vector2(-35, 5)
		label.z_index = 3
		_pile_layer.add_child(label)
		var tween := label.create_tween()
		tween.set_parallel(true)
		tween.tween_property(label, "position:y", label.position.y - 32, 0.6)
		tween.tween_property(label, "modulate:a", 0.0, 0.3).set_delay(0.3)
		tween.chain().tween_callback(label.queue_free)


func _compact_hp(value: float) -> String:
	if value >= 100000000:
		return ("%.1f" % (value / 100000000.0)).trim_suffix(".0") + "亿"
	if value >= 10000:
		return ("%.1f" % (value / 10000.0)).trim_suffix(".0") + "万"
	return str(roundi(value))


func _update_cards() -> void:
	for i in range(_cards.size()):
		_update_card(i)


# =====================================================================
# 输入：单击范围拍击 / 滚轮缩放 / 靠边移动；中键拖动作为附加操作
# =====================================================================
func _on_table_input(ev: InputEvent) -> void:
	if not _input_allowed(): return
	if ev is InputEventMouse and _over_camera_ui(ev.position): return
	if ev is InputEventMouseButton:
		var mb := ev as InputEventMouseButton
		if mb.button_index == MOUSE_BUTTON_WHEEL_UP or mb.button_index == MOUSE_BUTTON_WHEEL_DOWN:
			if mb.pressed:
				_set_camera_zoom(_camera_zoom * (1.15 if mb.button_index == MOUSE_BUTTON_WHEEL_UP else 1.0 / 1.15), mb.position)
			return
		if mb.button_index == MOUSE_BUTTON_MIDDLE:
			_panning = mb.pressed
			_update_slap_cursor(mb.position)
			return
		if mb.button_index == MOUSE_BUTTON_LEFT and mb.pressed and not _panning:
			_slap_area(mb.position)
		return
	if ev is InputEventMouseMotion:
		var mm := ev as InputEventMouseMotion
		if _panning:
			_camera_pan += mm.relative
			_apply_camera()
		_update_slap_cursor(mm.position)

func _area_targets(world_pos: Vector2, radius: float) -> Array:
	var hits: Array = []
	for i in range(mini(_cards.size(), GameState.battle.size())):
		var card: Dictionary = _cards[i]
		if bool(card.dead) or float(GameState.battle[i].hp) <= 0: continue
		var half: Vector2 = card.size * 0.5
		var center: Vector2 = card.get("rest_pos", card.node.position) + half
		var point := (world_pos - center).rotated(-float(card.rot))
		var closest := Vector2(clampf(point.x, -half.x, half.x), clampf(point.y, -half.y, half.y))
		if point.distance_squared_to(closest) <= (radius + 0.001) * (radius + 0.001):
			hits.append(i)
	return hits

func _slap_area(screen_pos: Vector2) -> void:
	if not _input_allowed() or not Rect2(Vector2.ZERO, _table.size).has_point(screen_pos): return
	if _over_camera_ui(screen_pos): return
	var world := _screen_to_world(screen_pos)
	var targets := _area_targets(world, GameState.slap_radius())
	if targets.is_empty() or not GameState.begin_slap_batch(): return
	var generation := GameState.battle_gen
	var gold_before := GameState.gold
	var damage_before := GameState.run_damage
	var count := 0
	for i in targets:
		if GameState.attack(int(i), false):
			count += 1
			if generation == GameState.battle_gen and not bool(_cards[i].dead):
				_shake(int(i), false)
	GameState.end_slap_batch()
	if count > 0:
		_spawn_area_impact(world, GameState.slap_radius(), count, GameState.gold - gold_before, GameState.run_damage - damage_before)
	_update_slap_cursor(screen_pos)

func _spawn_area_impact(world_pos: Vector2, radius: float, count: int, gold_gain: float, damage: float) -> void:
	var pulse := SlapCursor.Pulse.new()
	pulse.radius = radius
	pulse.position = world_pos
	_pile_layer.add_child(pulse)
	if is_instance_valid(_slap_receipt):
		_slap_receipt.visible = false
		_slap_receipt.queue_free()
	_slap_receipt = SlapFeedback.new()
	_slap_receipt.name = "SlapIncomeReceipt"
	_slap_receipt.damage_text = _fmt_pow(damage)
	_slap_receipt.gold_text = _fmt_pow(gold_gain)
	_slap_receipt.hit_count = count
	_slap_receipt.impact_radius = radius * _camera_scale
	_slap_receipt.position = _world_to_screen(world_pos)
	_table.add_child(_slap_receipt)

func _hit_index_at(local_pos: Vector2) -> int:
	local_pos = _screen_to_world(local_pos)
	var best := -1
	var best_d := 1e20
	for i in range(_cards.size()):
		var card: Dictionary = _cards[i]
		if card["dead"]:
			continue
		var node: Control = card["node"]
		if not node.visible:
			continue
		var sz: Vector2 = card["size"]
		# 在卡牌自身的（未旋转）坐标系里做判定 —— 不依赖引擎的变换栈
		var local: Vector2 = (local_pos - card.get("rest_pos", node.position) - sz * 0.5).rotated(-float(card.rot))
		if absf(local.x) <= sz.x * 0.5 and absf(local.y) <= sz.y * 0.5:
			# 重叠时取离牌心最近的那张（上层被压住的牌只算边角）
			var d: float = local.length()
			if d < best_d:
				best_d = d
				best = i
	return best


func _slap(i: int, heavy: bool, pos: Vector2) -> void:
	if not GameState.in_battle or i < 0 or i >= GameState.battle.size():
		return
	if float(GameState.battle[i]["hp"]) <= 0.0:
		return
	var generation := GameState.battle_gen
	if not GameState.attack(i, heavy):
		return
	# 清桌会自动补牌。新桌的卡不能承接上一桌的摇动和翻牌动画。
	_spawn_impact(pos, i, heavy)
	if generation == GameState.battle_gen and i < _cards.size() and i < GameState.battle.size() and not bool(_cards[i]["dead"]):
		_shake(i, heavy)
		if float(GameState.battle[i]["hp"]) <= 0.0:
			_flip_card(i)


func _shake(i: int, heavy: bool) -> void:
	var card: Dictionary = _cards[i]
	var node: Control = card["node"]
	var old_tween := card.get("shake_tween") as Tween
	if old_tween != null and old_tween.is_valid():
		old_tween.kill()
	var base: Vector2 = card.get("rest_pos", node.position)
	var amp := 7.0 if heavy else 3.0
	var tw := node.create_tween()
	card["shake_tween"] = tw
	tw.tween_property(node, "position", base + Vector2(amp, -amp * 0.4), 0.04)
	tw.tween_property(node, "position", base + Vector2(-amp * 0.7, amp * 0.3), 0.05)
	tw.tween_property(node, "position", base, 0.07)


func _spawn_impact(pos: Vector2, _i: int, heavy: bool) -> void:
	var info: Dictionary = GameState.last_slap
	var tag := str(info.get("tag", ""))
	var impact := InkImpact.new()
	impact.heavy = heavy
	impact.finisher = tag.contains("拍案")
	impact.set_meta("print_impact", true)
	_pile_layer.add_child(impact)
	impact.position = pos - impact.size * 0.5
	var txt := "-%s" % _fmt_pow(float(info.get("damage", 0.0)))
	var gain := float(info.get("gold_gain", 0.0))
	if gain > 0.0:
		txt += "  +%s金" % _fmt_pow(gain)
	if tag != "":
		txt = "%s %s" % [tag, txt]
	var lbl := Art.label(txt, 20 if heavy else 15,
		Art.GOLD if tag != "" else Art.RED if heavy else Art.INK)
	lbl.add_theme_color_override("font_outline_color", Art.PAPER_LIGHT)
	lbl.add_theme_constant_override("outline_size", 3)
	lbl.position = pos + Vector2(-18, -30)
	lbl.z_index = 2
	_pile_layer.add_child(lbl)
	var tw := lbl.create_tween()
	tw.set_parallel(true)
	tw.tween_property(lbl, "position", lbl.position + Vector2(0, -40), 0.5)
	tw.tween_property(lbl, "modulate:a", 0.0, 0.35).set_delay(0.15)
	tw.chain().tween_callback(lbl.queue_free)
	var stamp := PanelContainer.new()
	var ink := Art.RED if heavy else Art.INK
	var sb := Art.panel(Color(0, 0, 0, 0), 2, ink, 2)
	stamp.add_theme_stylebox_override("panel", sb)
	stamp.mouse_filter = Control.MOUSE_FILTER_IGNORE
	var stamp_size := Vector2(38, 38) if heavy else Vector2(28, 28)
	stamp.size = stamp_size
	stamp.position = pos - stamp_size * 0.5
	stamp.pivot_offset = stamp_size * 0.5
	stamp.rotation = -0.16
	var mark := Art.label("重" if heavy else "拍", 22 if heavy else 16, ink)
	mark.add_theme_font_override("font", Art.title_font())
	mark.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
	mark.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
	stamp.add_child(mark)
	_pile_layer.add_child(stamp)
	var st := stamp.create_tween()
	st.set_parallel(true)
	st.tween_property(stamp, "scale", Vector2(1.35, 1.35), 0.3)
	st.tween_property(stamp, "modulate:a", 0.0, 0.3)
	st.chain().tween_callback(stamp.queue_free)


func _flip_card(i: int) -> void:
	var card: Dictionary = _cards[i]
	if card["dead"]:
		return
	card["dead"] = true
	var node: Control = card["node"]
	var tw := node.create_tween()
	tw.tween_property(node, "scale:x", 0.04, 0.1)
	tw.tween_callback(func():
		for child in node.get_children():
			if child is CanvasItem:
				child.visible = false
		var back := TextureRect.new()
		back.texture = load("res://assets/art_v2/cardback.svg")
		back.expand_mode = TextureRect.EXPAND_IGNORE_SIZE
		back.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
		back.mouse_filter = Control.MOUSE_FILTER_IGNORE
		node.add_child(back)
		var seal := Art.label("拍", roundi(node.size.x * 0.22), Art.PAPER_LIGHT)
		seal.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
		seal.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
		seal.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
		back.add_child(seal))
	tw.tween_property(node, "scale:x", 1.0, 0.12)
	tw.tween_property(node, "position", node.position + Vector2(0, 60), 0.18)
	tw.parallel().tween_property(node, "scale", Vector2(0.35, 0.35), 0.18)
	tw.parallel().tween_property(node, "modulate:a", 0.0, 0.18)
	tw.tween_callback(func(): node.visible = false)

func _default_region() -> int:
	if GameState.in_battle:
		return GameState.battle_region
	if str(GameState.last_outcome.get("result", "")) == "settled":
		var previous := int(GameState.last_outcome.get("region", 0))
		if GameState.region_status(previous) == "available":
			return previous
	# 刚拍完的那一桌：优先把它刚解锁的邻居推出来，让玩家顺着线往下走
	var from := GameState.battle_region
	if from > 0:
		for n in GameData.region(from).get("unlocks", []):
			var j := int(n)
			if GameState.region_status(j) == "available":
				return j
	for r in GameData.regions:
		var i := int(r["idx"])
		if GameState.region_status(i) == "available":
			return i
	return GameState.START_REGION

func _refresh_map_panels() -> void:
	if _map_layer == null or not _map_layer.visible:
		return
	var total := GameState.total_field_regions()
	var done := GameState.cleared_count()
	_map_progress.text = "已克服  %d / %d" % [done, total]

	var parts := []
	for rt in GameState.ROUTES:
		var n := 0
		var c := 0
		for r in GameData.regions:
			if str(r.get("route", "")) == rt:
				n += 1
				if GameState.region_status(int(r["idx"])) == "cleared":
					c += 1
		parts.append("%s %d/%d" % [rt, c, n])
	_map_legend.text = "　".join(parts)

	_map_foot.text = "亮起的节点可以立刻开打　·　点节点看详情，点「铺这一桌」把牌铺到桌面上" \
		+ "　·　当前：%s" % (_region_brief(_sel_idx) if _sel_idx > 0 else "未选")

	_fill_map_left()
	_fill_map_right()


func _region_brief(idx: int) -> String:
	if idx <= 0:
		return "—"
	var r := GameData.region(idx)
	return "%s（%s）" % [r.get("name", "?"), r.get("route", "")]

func _append_log(t: String) -> void:
	_log_lines.append(t)
	while _log_lines.size() > LOG_MAX:
		_log_lines.pop_front()
	_render_log()


func _render_log() -> void:
	if _log_lbl == null:
		return
	_log_lbl.text = "　·　".join(_log_lines)


func _join_str(arr: Array, sep: String) -> String:
	var s := ""
	for i in range(arr.size()):
		if i > 0:
			s += sep
		s += str(arr[i])
	return s


# =====================================================================
# 底部上阵条：拖拽源 / 落点（v1.0）
# =====================================================================
## ⚠️ 用**内部类**而不是新建 class_name 脚本 —— 新 class_name 会写进全局类缓存，
##    没跑 `--import` 就直接报 `Could not find type`（本项目已经踩过两次）。
## ⚠️ 这些控件的**子 Label 必须 mouse_filter = IGNORE**：Godot 只问鼠标下**最上面那个**
##    Control 要不要 `_get_drag_data()`，不会往上冒泡 —— 子 Label 吃掉了按下事件，
##    父容器就永远收不到拖拽起手。

## 卡池里的一张将兵牌：拖出去 → 落到槽位上阵。
class _PoolCard extends PanelContainer:
	# ⚠️ host **不能**标注成 Control：那样分析器会在 `host._drag_preview()` 上报
	#    "Function not found in base Control"，因为它只知道 Control 的成员。
	#    留成 Variant 走动态派发，才能在运行时调到父脚本 main.gd 的方法。
	var host = null                     # 即 main.gd（父脚本）
	var card_id := ""

	func _get_drag_data(_at: Vector2) -> Variant:
		if card_id == "" or host == null:
			return null
		# ⚠️ set_drag_preview() 只在**真实拖拽会话**里合法，否则引擎会打
		#    "Condition !get_viewport()->gui_is_dragging() is true" 的 ERROR。
		#    自检（无头）会直接调 _get_drag_data() 来验 payload —— 那时没有拖拽会话，
		#    所以必须先问一句。（已实测：真实拖拽时 gui_is_dragging() 确实是 true，
		#    所以这个守卫不会让预览消失。）
		var vp := get_viewport()
		if vp != null and vp.gui_is_dragging():
			set_drag_preview(host._drag_preview(card_id))
		# 用带 kind 的字典而不是裸 id：防止"把卡池牌拖到卡池上"被当成一次成功操作
		return {"paan_kind": "pool", "id": card_id}


## 一个上阵槽位。收卡池来的牌；左键点一下 = 卸下。主角位 locked，两点都不响应。
class _Slot extends PanelContainer:
	var host = null                     # main.gd（同上，必须留 Variant）
	var card_id := ""
	var slot_index := -1
	var locked := false

	func _can_drop_data(_at: Vector2, data: Variant) -> bool:
		if locked or host == null:
			return false
		return data is Dictionary and str((data as Dictionary).get("paan_kind", "")) == "pool"

	func _drop_data(_at: Vector2, data: Variant) -> void:
		host._deploy_at(str((data as Dictionary).get("id", "")), slot_index)

	func _gui_input(ev: InputEvent) -> void:
		if locked or card_id == "":
			return
		if ev is InputEventMouseButton:
			var mb := ev as InputEventMouseButton
			if mb.pressed and mb.button_index == MOUSE_BUTTON_LEFT:
				host._undeploy(card_id)
				accept_event()


class EmptySlotOutline extends Control:
	func _ready() -> void:
		mouse_filter = Control.MOUSE_FILTER_IGNORE
		set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
		resized.connect(queue_redraw)
	func _draw() -> void:
		var rect := Rect2(Vector2(5, 5), size - Vector2(10, 10))
		var color := Color("a18c6b")
		for pair in [[rect.position, Vector2(rect.end.x, rect.position.y)],
			[Vector2(rect.end.x, rect.position.y), rect.end],
			[rect.end, Vector2(rect.position.x, rect.end.y)],
			[Vector2(rect.position.x, rect.end.y), rect.position]]:
			draw_dashed_line(pair[0], pair[1], color, 1.0, 4.0)
